Código de Conexión para tu Juego

Copia este código y colócalo en tu HTML para guardar inicios de sesión, posiciones o datos.


// Importa el CDN de Hitbase
<script src="https://cdn.makebyr.github.io/hit.base/v2/hitbase.js"></script>

<script>
  const hitbaseConfig = {
      apiKey: "HITBASE_API_KEY_AQUI",
      serverDomain: "hit./HITBASE_API_KEY_AQUI"
  };
  Hitbase.initializeApp(hitbaseConfig);

  // Instancia de Realtime Database
  const create_file_realtimedatabase = Hitbase.database();

  // Ejemplo: Guardar movimiento de un jugador
  function guardarMovimiento(jugadorId, posX, posY) {
      create_file_realtimedatabase.ref('jugadores/' + jugadorId).set({
          x: posX,
          y: posY,
          ultimaActualizacion: Date.now()
      });
  }
</script>
